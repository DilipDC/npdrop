    df["Age"] = pd.to_numeric(df["Age"], errors="coerce")
    df["Age"] = df["Age"].fillna(df["Age"].median())

    print(df["Region"].value_counts(dropna=False))

# IQR outlier replacement with mean (example on numeric column)
col = df.select_dtypes(include=[np.number]).columns[0]
q1 = df[col].quantile(0.25)
q3 = df[col].quantile(0.75)
iqr = q3 - q1
lower_iqr = q1 - 1.5 * iqr
upper_iqr = q3 + 1.5 * iqr
mean_val = df[col].mean()
df[col] = df[col].mask((df[col] < lower_iqr) | (df[col] > upper_iqr), mean_val)

print("Lemmatized:", [wl.lemmatize(w, pos="v") for w in tokens])

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

df = pd.read_csv("Mall_Customers.csv")
X = df[["Annual Income (k$)", "Spending Score (1-100)"]]

# Elbow Method (WCSS)
wcss = []
for k in range(1, 11):
    km = KMeans(n_clusters=k, random_state=42, n_init="auto")
    km.fit(X)
    wcss.append(km.inertia_)

plt.figure(figsize=(7, 4))
sns.lineplot(x=range(1, 11), y=wcss, marker="o")
plt.title("Elbow Method for Optimal k")
plt.xlabel("Number of Clusters (k)")
plt.ylabel("WCSS (Inertia)")
plt.xticks(range(1, 11))
plt.show()

# Fit K-Means with k = 5
model = KMeans(n_clusters=5, random_state=42, n_init="auto")
df["cluster"] = model.fit_predict(X)

plt.figure(figsize=(8, 5))
sns.scatterplot(
    data=df,
    x="Annual Income (k$)",
    y="Spending Score (1-100)",
    hue="cluster",
    palette="viridis",
    s=60
)
centroids = model.cluster_centers_
plt.scatter(
    centroids[:, 0],
    centroids[:, 1],
    color="red",
    marker="X",
    s=200,
    label="Centroids"
)
plt.title("Customer Segments (K-Means, k=5)")
plt.xlabel("Annual Income (k$)")
plt.ylabel("Spending Score (1-100)")
plt.legend()
plt.show()

from functools import reduce
import pandas as pd

nums = [1, 2, 3, 4, 5, 6]

# 1. lambda (small anonymous function)
add_ten = lambda x: x + 10
print(add_ten(5))  # 15

# 2. map() (apply function to every item -> wrap in list())
squares = list(map(lambda x: x ** 2, nums))
print(squares)  # [1, 4, 9, 16, 25, 36]

# 3. filter() (keep items where condition is True -> wrap in list())
evens = list(filter(lambda x: x % 2 == 0, nums))
print(evens)  # [2, 4, 6]

# 4. reduce() (fold sequence into a single value)
total = reduce(lambda a, b: a + b, nums)
print(total)  # 21

# 5. List Comprehension (clean Python alternative to map + filter)
even_squares = [x ** 2 for x in nums if x % 2 == 0]
print(even_squares)  # [4, 16, 36]

# 6. fillna() (replace missing/NaN values in Pandas)
df = pd.DataFrame({"Score": [80.0, None, 90.0, None], "Grade": ["A", None, "B", "A"]})
df["Score"] = df["Score"].fillna(df["Score"].mean())   # fill numeric NaNs with mean/median
df["Grade"] = df["Grade"].fillna(df["Grade"].mode()[0]) # fill categorical NaNs with mode
print(df)

# 7. Pandas Series.map() & apply() (value mapping & custom row/cell logic)
df["Pass"] = df["Score"].map(lambda s: 1 if s >= 85 else 0)
print(df)
